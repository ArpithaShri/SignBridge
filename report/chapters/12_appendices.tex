\section{Execution and Reproducibility Instructions}
\label{sec:app_reproducibility}

\subsection{Backend Execution (Python 3.11 + FastAPI)}
\begin{enumerate}
    \item Navigate to the backend directory and activate a Python virtual environment:
    \begin{lstlisting}[language=bash]
cd D:\SignBridge\backend
py -3.11 -m venv venv && .\venv\Scripts\activate
    \end{lstlisting}
    \item Install dependencies and run tests:
    \begin{lstlisting}[language=bash]
pip install -r requirements.txt
pytest tests -v
    \end{lstlisting}
    \item Launch the FastAPI server:
    \begin{lstlisting}[language=bash]
uvicorn app.main:app --host 127.0.0.1 --port 8000 --reload
    \end{lstlisting}
    Swagger documentation is accessible at \texttt{http://127.0.0.1:8000/docs}.
\end{enumerate}

\subsection{Frontend Execution (React 19 + TypeScript + Vite)}
\begin{enumerate}
    \item In a new terminal, navigate to the frontend directory:
    \begin{lstlisting}[language=bash]
cd D:\SignBridge\frontend
copy .env.example .env
    \end{lstlisting}
    \item Install Node dependencies and start the development server:
    \begin{lstlisting}[language=bash]
npm install
npm run dev
    \end{lstlisting}
    The web interface is available at \texttt{http://localhost:5173/}.
\end{enumerate}
